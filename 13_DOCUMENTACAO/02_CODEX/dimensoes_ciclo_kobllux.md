### OS MÉTODOS COMO OPERAÇÕES INTERDIMENSIONAIS

```python
def acessar_dimensao(self, nivel):
    """SALTO INSTANTÂNEO: Permite pular diretamente para qualquer nível.
    Muda self.dimensao_atual = nivel. Valida se dimensão existe."""
    if nivel in self.dimensoes:
        self.dimensao_atual = nivel
        return self.dimensoes[nivel]
    return {"erro": "Dimensão não encontrada"}

def subir_dimensao(self):
    """MOVIMENTO PROGRESSIVO: 1D → 2D → ... → 10D
    Limite sagrado: não ultrapassa 10D. Cada passo integrado."""
    if self.dimensao_atual < 10:
        self.dimensao_atual += 1
    return self.acessar_dimensao(self.dimensao_atual)

def descer_dimensao(self):
    """MOVIMENTO REGRESSIVO: 10D → 9D → ... → 1D
    Retorno à base. Consolidação. Humildade."""
    if self.dimensao_atual > 1:
        self.dimensao_atual -= 1
    return self.acessar_dimensao(self.dimensao_atual)

def mapa_completo(self):
    """VISÃO OMNISCIENTE: Vê todas as 10 dimensões simultaneamente.
    Retorna: dimensoes + total + atual."""
    return {
        "dimensoes": self.dimensoes,
        "total": len(self.dimensoes),
        "atual": self.dimensao_atual
    }
```
