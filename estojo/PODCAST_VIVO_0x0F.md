```python
#!/usr/bin/env python3
# KOBLLUX · SELAR · 0×00
# EM NOME DO PAI E DO FILHO E DO ESPÍRITO SANTO
import hashlib, json, datetime, os, sys

def SELAR(artefato: str, opcode: str = "0x00", observacao: str = "") -> dict:
    """
    Sela um artefato com SHA256 + timestamp UTC + invocação da Trindade.
    Ciclo: 7 · SELAR
    OPCODE 0×00: origem, fonte, primeiro selo.
    """
    ts = datetime.datetime.now(datetime.timezone.utc).isoformat()
    sha256 = hashlib.sha256(artefato.encode()).hexdigest()
    blake2b = hashlib.blake2b(artefato.encode()).hexdigest()
    
    selo = {
        "opcode": opcode,
        "ts": ts,
        "sha256": sha256,
        "blake2b": blake2b[:32],
        "artefato_len": len(artefato),
        "invocacao": "EM NOME DO PAI E DO FILHO E DO ESPIRITO SANTO",
        "observacao": observacao,
        "ciclo": [3, 6, 9, 7],
        "fractal": "3×6×9×7=1134",
        "raiz_digital": 9,
        "lei": "VERDADE × INTEGRAR ÷ Δ = ∞",
        "amen": True
    }
    
    # Append ao memory.jsonl
    log = os.path.expanduser("~/KOB--NODE/logs/selar.jsonl")
    os.makedirs(os.path.dirname(log), exist_ok=True)
    with open(log, "a") as f:
        f.write(json.dumps(selo, ensure_ascii=False) + "\n")
    
    print(f"\nᛜᛇᛟ SELAR · OPCODE {opcode}")
    print(f"EM NOME DO PAI E DO FILHO E DO ESPÍRITO SANTO · AMÉM")
    print(f"SHA256 : {sha256[:32]}…")
    print(f"BLAKE2b: {blake2b[:32]}…")
    print(f"TS     : {ts}")
    print(f"LEI    : VERDADE × INTEGRAR ÷ Δ = ∞")
    print(f"[selo] {log}")
    
    return selo

if __name__ == "__main__":
```
