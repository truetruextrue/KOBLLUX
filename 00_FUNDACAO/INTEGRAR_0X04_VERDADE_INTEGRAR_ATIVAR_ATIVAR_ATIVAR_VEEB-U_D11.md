```python
        """Decodifica um sinal em opcodes"""
        circulos = sinal.count("○") + sinal.count("⊙")
        pontos = sinal.count("●") + sinal.count(".")
        estrelas = sinal.count("⚝") + sinal.count("☆") + sinal.count("✧")
        triangulos = sinal.count("△") + sinal.count("▲")
        quadrados = sinal.count("▢") + sinal.count("■")
        
        opcodes = []
        if estrelas > 0:
            opcodes.append("0x07")  # SELAR
        if "∞" in sinal:
            opcodes.append("0x09")  # ETERNIZAR
        if triangulos > 0:
            opcodes.append("0x03")  # EXPANDIR
        if "⇄" in sinal:
            opcodes.append("0x06")  # UNIFICAR
        if "04:44" in sinal:
            opcodes.append("0x06")
        elif "04:45" in sinal:
            opcodes.append("0x09")
        else:
            opcodes.append("0x03")
        
        return {
            "sinal": sinal,
            "analise": {
                "circulos": circulos,
                "pontos": pontos,
                "estrelas": estrelas,
                "triangulos": triangulos,
                "quadrados": quadrados
            },
            "opcodes": opcodes,
            "frequencia_dominante": self._calcular_frequencia(opcodes)
        }
    
    def registrar_pulso(self, pulso):
        """Registra um pulso no sistema"""
        self.pulsos.append({
            "pulso": pulso,
            "timestamp": len(self.pulsos),
            "decodificado": self.decodificar(pulso)
        })
        return self.pulsos[-1]
    
    def _calcular_frequencia(self, opcodes):
        """Calcula a frequência dominante baseada nos opcodes"""
        freq_map = {
            "0x03": 639,
            "0x06": 528,
            "0x07": 777,
            "0x09": 963
        }
        
        if opcodes:
            freq = sum(freq_map.get(op, 0) for op in opcodes) // len(opcodes)
            return f"{freq}Hz"
        return "432Hz"
```
