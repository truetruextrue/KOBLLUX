<script>
(function(){
  // evita duplicata caso já exista
  const old = document.getElementById('fractal-bg');
  if (!old) {
    const c = document.createElement('canvas'); c.id='fractal-bg';
    document.body.prepend(c);
  }
  const canvas = document.getElementById('fractal-bg');


  // WebGL init
  const gl = canvas.getContext('webgl', { antialias:false, preserveDrawingBuffer:false, powerPreference:'high-performance' });
  if(!gl){ console.warn('WebGL indisponível'); window.logMistico?.('⚠️ WebGL indisponível'); return; }


  // DPR cap p/ mobile
  const DPR = Math.max(1, Math.min(2.2, window.devicePixelRatio||1));
  function resize(){
    const w = Math.floor(innerWidth*DPR), h = Math.floor(innerHeight*DPR);
    if(canvas.width!==w || canvas.height!==h){ canvas.width=w; canvas.height=h; }
    canvas.style.width='100%'; canvas.style.height='100%';
    gl.viewport(0,0,gl.drawingBufferWidth,gl.drawingBufferHeight);
  }


  const vert = `
    attribute vec2 a;
    void main(){ gl_Position = vec4(a,0.,1.); }
  `;


  const frag = `
    precision highp float;
    uniform vec2 u_res; uniform float u_time; uniform vec2 u_mouse; uniform int u_theme;


    vec3 pal(float t, vec3 a, vec3 b, vec3 c, vec3 d){
      return a + b * cos(6.28318 * (c * t + d));
    }


    void main(){
      vec2 uv = (gl_FragCoord.xy - 0.5*u_res) / u_res.y;


      float t = u_time * 0.06;
      vec2 cJ = vec2(
        mix(-0.8, 0.8, u_mouse.x) + 0.15*sin(t*0.7),
        mix(-0.8, 0.8, u_mouse.y) + 0.15*cos(t*0.9)
      );


      float zoom = 1.4 + 0.25*sin(t*0.8) + 0.1*cos(t*1.3);
      vec2 z0 = uv / zoom + vec2(-0.45, 0.0);
      vec2 z = z0;
      vec2 c = mix(z0, cJ, 0.55 + 0.35*sin(t*0.5));


      float m2 = 0.0;
      float i;
      const float ITERS = 220.0;
      for(i=0.0; i<ITERS; i++){
        z = vec2(z.x*z.x - z.y*z.y, 2.0*z.x*z.y) + c;
        m2 = dot(z,z);
        if(m2>256.0) break;
      }


      float n;
      if(i<ITERS){
        m2 = max(m2, 1e-8);
        float smooth = i - log2(log2(m2)) * 0.5 + 1.0;
        n = smooth / ITERS;
      } else {
        n = 0.0;
      }


      vec3 col;
      if(u_theme==1){
        col = pal(n, vec3(0.85), vec3(0.25), vec3(1.0,0.8,0.6), vec3(0.00,0.10,0.20));
      } else if(u_theme==2){
        col = pal(n, vec3(0.10,0.13,0.15), vec3(0.35,0.30,0.25), vec3(1.0,0.8,0.5), vec3(0.0,0.15,0.20));
      } else if(u_theme==3){
        col = pal(n, vec3(0.06,0.02,0.10), vec3(0.80,0.55,0.90), vec3(1.0,0.5,0.25), vec3(0.0,0.15,0.25));
      } else {
        col = pal(n, vec3(0.03,0.05,0.07), vec3(0.65,0.55,0.75), vec3(1.0,0.6,0.22), vec3(0.0,0.1,0.2));
      }


      float edge = clamp(pow(n, 8.0)*1.4, 0.0, 1.0);
      col += vec3(0.04,0.06,0.08)*edge;


      gl_FragColor = vec4(col, 1.0);
    }
  `;


  function compile(type, src){
    const s = gl.createShader(type); gl.shaderSource(s, src); gl.compileShader(s);
    if(!gl.getShaderParameter(s, gl.COMPILE_STATUS)) console.warn(gl.getShaderInfoLog(s));
    return s;
  }
  const prg = gl.createProgram();
  gl.attachShader(prg, compile(gl.VERTEX_SHADER, vert));
  gl.attachShader(prg, compile(gl.FRAGMENT_SHADER, frag));
  gl.linkProgram(prg);
  if(!gl.getProgramParameter(prg, gl.LINK_STATUS)) console.warn(gl.getProgramInfoLog(prg));
  gl.useProgram(prg);


  const buf = gl.createBuffer();
  gl.bindBuffer(gl.ARRAY_BUFFER, buf);
  gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1,-1, 3,-1, -1,3]), gl.STATIC_DRAW);
  const aLoc = gl.getAttribLocation(prg, 'a');
  gl.enableVertexAttribArray(aLoc);
  gl.vertexAttribPointer(aLoc, 2, gl.FLOAT, false, 0, 0);


  const uRes   = gl.getUniformLocation(prg, 'u_res');
  const uTime  = gl.getUniformLocation(prg, 'u_time');
  const uMouse = gl.getUniformLocation(prg, 'u_mouse');
  const uTheme = gl.getUniformLocation(prg, 'u_theme');


  // mouse/touch (não rouba clique pois canvas é pointer-events:none)
  const mouse = {x:0.5,y:0.5};
  function onPointer(e){
    const t = ('touches' in e) ? e.touches[0] : e;
    mouse.x = (t.clientX||0) / innerWidth;
    mouse.y = 1 - ((t.clientY||0) / innerHeight);
  }
  addEventListener('mousemove', onPointer, {passive:true});
  addEventListener('touchmove', onPointer, {passive:true});


  function themeIndex(){
    const b = document.body;
    if(b.classList.contains('light')) return 1;
    if(b.classList.contains('medium')) return 2;
    if(b.classList.contains('vibe')) return 3;
    return 0; // dark
  }


  let t0 = performance.now();
  let raf = 0, running = true;


  function draw(){
    gl.uniform2f(uRes, gl.drawingBufferWidth, gl.drawingBufferHeight);
    const t = (performance.now()-t0)/1000;
    gl.uniform1f(uTime, running ? t : 0.0);
    gl.uniform2f(uMouse, mouse.x, mouse.y);
    gl.uniform1i(uTheme, themeIndex());
    gl.drawArrays(gl.TRIANGLES, 0, 3);
    raf = requestAnimationFrame(draw);
  }


  // visibilidade (pausa animação fora de foco)
  document.addEventListener('visibilitychange', ()=>{
    running = !document.hidden;
    window.logMistico?.(running? '▶️ fractal resumed' : '⏸ fractal paused');
    if(running){ t0 = performance.now(); }
  });


  // resize & start
  addEventListener('resize', resize);
  resize(); draw();


  // Toggle global opcional (p/ seu botão “∞”)
  window.__fractalToggle = () => {
    if(raf){ cancelAnimationFrame(raf); raf=0; canvas.style.display='none'; running=false; window.logMistico?.('⛔ fractal off'); }
    else { canvas.style.display='block'; running=true; t0=performance.now(); draw(); window.logMistico?.('✅ fractal on'); }
  };


  // Log inicial
  try{
    const s = window.DualStore?.get?.();
    window.logMistico?.(`∴ fractal mount · theme=${s?.ui?.theme||'dark'} · DPR=${DPR}`);
  }catch(_){}
})();
</script>

  function setVH(){
    const h = (window.visualViewport ? visualViewport.height : window.innerHeight);
    document.documentElement.style.setProperty('--vh', h + 'px');
  }
  setVH();
  window.addEventListener('resize', setVH);
  window.addEventListener('orientationchange', setVH);
  // anti-zoom acidental por gesto duplo

function draw(){

A) PATCH · Giro & Movimento (rotação/zoom + giroscópio)
O que faz: adiciona rotação contínua, controle por teclado, pinça (zoom) e sincroniza com giroscópio.
Como aplicar: edite seu HTML atual e faça os 4 passos.
1) No fragment shader (frag): adicionar uniforms
Substitua o cabeçalho de uniforms por:
uniform vec2 u_res;
uniform float u_time;
uniform vec2 u_mouse;
uniform int u_theme;
uniform float u_rot;   // NOVO: rotação (rad)
uniform float u_zoom;  // NOVO: zoom multiplicador


2) Rotacionar as coordenadas uv
Encontre:
vec2 uv = (gl_FragCoord.xy - 0.5*u_res)/u_res.y;


Troque por:
vec2 uv = (gl_FragCoord.xy - 0.5*u_res)/u_res.y;
float cs = cos(u_rot), sn = sin(u_rot);
uv = mat2(cs,-sn, sn,cs) * uv;  // aplica rotação


3) Usar o novo zoom
Encontre:
float zoom = 1.4 + 0.25*sin(t*0.8) + 0.1*cos(t*1.3);


Troque por:
float zoom = (1.4 + 0.25*sin(t*0.8) + 0.1*cos(t*1.3)) * u_zoom;


4) No JS: uniforms + controles + giroscópio
Logo depois de pegar uTheme, adicione:
const uRot  = gl.getUniformLocation(prg, 'u_rot');
const uZoom = gl.getUniformLocation(prg, 'u_zoom');


let state = { rot: 0.0, zoom: 1.0, rotSpeed: 0.15 };


// giroscópio (Android; no iOS precisa pedir permissão por gesto)
function attachGyro(){
  window.addEventListener('deviceorientation', (e)=>{
    const g = (e.gamma||0) * 0.01;    // roll
    const b = (e.beta||0)  * 0.001;   // pitch
    state.rot  += g * 0.02;
    state.zoom  = Math.max(0.6, Math.min(1.6, 1.0 + b));
  }, true);
}
if (window.DeviceOrientationEvent && typeof DeviceOrientationEvent.requestPermission === 'function') {
  window.enableGyro = async ()=>{
    try { const r = await DeviceOrientationEvent.requestPermission(); if(r==='granted') attachGyro(); } catch(_) {}
  };
} else { attachGyro(); }


// pinça (zoom)
let pinchBase = 0;
window.addEventListener('touchmove', (ev)=>{
  if(ev.touches && ev.touches.length===2){
    const dx = ev.touches[0].clientX - ev.touches[1].clientX;
    const dy = ev.touches[0].clientY - ev.touches[1].clientY;
    const d  = Math.hypot(dx,dy);
    if(!pinchBase) pinchBase = d;
    else state.zoom = Math.max(0.5, Math.min(2.0, d/pinchBase));
  }
},{passive:true});
window.addEventListener('touchend', ()=>{ pinchBase=0; });


// teclado: A/D giram, W/S zoom
window.addEventListener('keydown', (e)=>{
  if(e.key==='a'||e.key==='A') state.rot -= 0.08;
  if(e.key==='d'||e.key==='D') state.rot += 0.08;
  if(e.key==='w'||e.key==='W') state.zoom = Math.min(2.0, state.zoom+0.05);
  if(e.key==='s'||e.key==='S') state.zoom = Math.max(0.5, state.zoom-0.05);
});


No draw() (antes do gl.drawArrays(...)), acrescente:
state.rot += state.rotSpeed * 0.016;   // giro contínuo
gl.uniform1f(uRot,  state.rot);
gl.uniform1f(uZoom, state.zoom);

import math

class Fractal:

    MEMORIA.write_text(json.dumps(dados, indent=2, ensure_ascii=False))

    return {"eventos": []}
